SELECT name
FROM user_profiles
WHERE name LIKE '%am%';

SELECT MAX(e.totalCost) AS max_expenses
FROM expenses e
JOIN cars c ON e.carId = c.id
JOIN car_models cm ON c.carModelId = cm.id
JOIN car_brands cb ON cm.carbrandId = cb.id
WHERE cb.title = 'Audi';
--------------------------------------------------------------------------------------------------

SELECT COUNT(cm.id) AS count_models, cb.id
FROM car_brands cb
JOIN car_models cm ON cm.carBrandId = cb.id
WHERE cb.title IN ('Audi', 'BMW')
GROUP BY cb.id;
--------------------------------------------------------------------------------------------------

SELECT cm.title AS car_model, cb.title AS car_brand, COUNT(c.userId) AS user_count
FROM cars c
JOIN car_models cm ON c.carModelId = cm.id
JOIN car_brands cb ON cm.carbrandId = cb.id
GROUP BY cm.id, cm.title, cb.id, cb.title;
--------------------------------------------------------------------------------------------------

SELECT up.name
FROM user_profiles up
JOIN cars c ON c.userId = up.id;
